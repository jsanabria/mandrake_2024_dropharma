/* ============================================================
   VISTAS CxP / CxC
   Blindaje de COLLATION
   Collation estándar utilizada: utf8mb4_unicode_ci
   ============================================================ */

SET NAMES utf8mb4 COLLATE utf8mb4_unicode_ci;
SET collation_connection = 'utf8mb4_unicode_ci';


/* ============================================================
   ELIMINAR PRIMERO LAS VISTAS DEPENDIENTES
   ============================================================ */

DROP VIEW IF EXISTS view_cuentas_por_pagar_resumen;
DROP VIEW IF EXISTS view_cuentas_por_cobrar_resumen;
DROP VIEW IF EXISTS view_cuentas_por_pagar;
DROP VIEW IF EXISTS view_cuentas_por_cobrar;


/* ============================================================
   1. CUENTAS POR PAGAR
   ============================================================ */

CREATE VIEW view_cuentas_por_pagar AS
SELECT
    base.id,
    base.proveedor,
    base.proveedor_rif,
    base.proveedor_nombre,

    base.tipo_documento,
    base.documento,
    base.nro_control,
    base.fecha,
    base.fecha_registro,
    base.descripcion,
    base.doc_afectado,
    base.anulado,
    base.pagado,

    base.moneda,
    base.tasa_dia,

    base.signo_documento,

    /* Monto sin signo */
    base.monto_documento_moneda,
    base.monto_documento_bs,
    base.monto_documento_usd,

    /* Monto con efecto fiscal */
    ROUND(
        base.monto_documento_bs * base.signo_documento,
        2
    ) AS monto_aplicado_bs,

    ROUND(
        base.monto_documento_usd * base.signo_documento,
        2
    ) AS monto_aplicado_usd,

    /* Pagos */
    IFNULL(base.total_pagado_bs, 0) AS total_pagado_bs,
    IFNULL(base.total_pagado_usd, 0) AS total_pagado_usd,
    IFNULL(base.cantidad_pagos, 0) AS cantidad_pagos,
    base.fecha_ultimo_pago,

    /* Saldos */
    ROUND(
        (base.monto_documento_bs * base.signo_documento)
        - IFNULL(base.total_pagado_bs, 0),
        2
    ) AS saldo_bs,

    ROUND(
        (base.monto_documento_usd * base.signo_documento)
        - IFNULL(base.total_pagado_usd, 0),
        2
    ) AS saldo_usd,

    /* Estado de cuenta */
    (
        CASE
            WHEN
                base.anulado COLLATE utf8mb4_unicode_ci
                =
                _utf8mb4'S' COLLATE utf8mb4_unicode_ci
            THEN _utf8mb4'ANULADO'

            WHEN
                base.tipo_documento COLLATE utf8mb4_unicode_ci
                =
                _utf8mb4'NC' COLLATE utf8mb4_unicode_ci
            THEN _utf8mb4'NOTA DE CRÉDITO'

            WHEN ROUND(
                (base.monto_documento_bs * base.signo_documento)
                - IFNULL(base.total_pagado_bs, 0),
                2
            ) <= 0.01
            THEN _utf8mb4'PAGADO'

            WHEN IFNULL(base.total_pagado_bs, 0) > 0
            THEN _utf8mb4'PAGO PARCIAL'

            ELSE _utf8mb4'PENDIENTE'
        END
    ) COLLATE utf8mb4_unicode_ci AS estado_cuenta,

    DATEDIFF(
        CURDATE(),
        base.fecha
    ) AS dias_transcurridos,

    /* Antigüedad */
    (
        CASE
            WHEN
                base.tipo_documento COLLATE utf8mb4_unicode_ci
                =
                _utf8mb4'NC' COLLATE utf8mb4_unicode_ci
            THEN _utf8mb4'NOTA DE CRÉDITO'

            WHEN DATEDIFF(CURDATE(), base.fecha) <= 0
            THEN _utf8mb4'NO VENCIDO'

            WHEN DATEDIFF(CURDATE(), base.fecha)
                 BETWEEN 1 AND 30
            THEN _utf8mb4'1-30 DÍAS'

            WHEN DATEDIFF(CURDATE(), base.fecha)
                 BETWEEN 31 AND 60
            THEN _utf8mb4'31-60 DÍAS'

            WHEN DATEDIFF(CURDATE(), base.fecha)
                 BETWEEN 61 AND 90
            THEN _utf8mb4'61-90 DÍAS'

            ELSE _utf8mb4'MÁS DE 90 DÍAS'
        END
    ) COLLATE utf8mb4_unicode_ci AS antiguedad

FROM
(
    SELECT
        c.id,
        c.proveedor,

        p.ci_rif COLLATE utf8mb4_unicode_ci
            AS proveedor_rif,

        p.nombre COLLATE utf8mb4_unicode_ci
            AS proveedor_nombre,

        c.tipo_documento COLLATE utf8mb4_unicode_ci
            AS tipo_documento,

        c.documento COLLATE utf8mb4_unicode_ci
            AS documento,

        c.nro_control COLLATE utf8mb4_unicode_ci
            AS nro_control,

        c.fecha,
        c.fecha_registro,

        c.descripcion COLLATE utf8mb4_unicode_ci
            AS descripcion,

        c.doc_afectado COLLATE utf8mb4_unicode_ci
            AS doc_afectado,

        c.anulado COLLATE utf8mb4_unicode_ci
            AS anulado,

        c.pagado COLLATE utf8mb4_unicode_ci
            AS pagado,

        c.moneda COLLATE utf8mb4_unicode_ci
            AS moneda,

        IFNULL(
            NULLIF(c.tasa_dia, 0),
            1
        ) AS tasa_dia,

        CASE
            WHEN
                c.tipo_documento COLLATE utf8mb4_unicode_ci
                =
                _utf8mb4'NC' COLLATE utf8mb4_unicode_ci
            THEN -1
            ELSE 1
        END AS signo_documento,

        /* ----------------------------------------------------
           Monto en moneda original
           ---------------------------------------------------- */

        CASE
            WHEN IFNULL(c.monto_pagar, 0) > 0
                THEN c.monto_pagar
            ELSE IFNULL(c.monto_total, 0)
        END AS monto_documento_moneda,

        /* ----------------------------------------------------
           Monto convertido a bolívares
           ---------------------------------------------------- */

        CASE
            WHEN
                UPPER(
                    LEFT(
                        TRIM(c.moneda COLLATE utf8mb4_unicode_ci),
                        3
                    )
                )
                =
                _utf8mb4'BS.' COLLATE utf8mb4_unicode_ci
            THEN
                CASE
                    WHEN IFNULL(c.monto_pagar, 0) > 0
                        THEN c.monto_pagar
                    ELSE IFNULL(c.monto_total, 0)
                END
            ELSE
                (
                    CASE
                        WHEN IFNULL(c.monto_pagar, 0) > 0
                            THEN c.monto_pagar
                        ELSE IFNULL(c.monto_total, 0)
                    END
                )
                *
                IFNULL(
                    NULLIF(c.tasa_dia, 0),
                    1
                )
        END AS monto_documento_bs,

        /* ----------------------------------------------------
           Monto convertido a USD
           ---------------------------------------------------- */

        CASE
            WHEN
                UPPER(
                    LEFT(
                        TRIM(c.moneda COLLATE utf8mb4_unicode_ci),
                        3
                    )
                )
                =
                _utf8mb4'BS.' COLLATE utf8mb4_unicode_ci
            THEN
                (
                    CASE
                        WHEN IFNULL(c.monto_pagar, 0) > 0
                            THEN c.monto_pagar
                        ELSE IFNULL(c.monto_total, 0)
                    END
                )
                /
                NULLIF(c.tasa_dia, 0)

            ELSE
                CASE
                    WHEN IFNULL(c.monto_pagar, 0) > 0
                        THEN c.monto_pagar
                    ELSE IFNULL(c.monto_total, 0)
                END
        END AS monto_documento_usd,

        pagos.total_pagado_bs,
        pagos.total_pagado_usd,
        pagos.cantidad_pagos,
        pagos.fecha_ultimo_pago

    FROM compra AS c

    LEFT JOIN proveedor AS p
        ON p.id = c.proveedor

    LEFT JOIN
    (
        SELECT
            pago_documento.id_documento,

            ROUND(
                SUM(pago_documento.pagado_bs),
                2
            ) AS total_pagado_bs,

            ROUND(
                SUM(pago_documento.pagado_usd),
                2
            ) AS total_pagado_usd,

            COUNT(*) AS cantidad_pagos,

            MAX(pago_documento.fecha) AS fecha_ultimo_pago

        FROM
        (
            SELECT
                pc.id,
                pc.id_documento,
                pc.fecha,

                /* Pago en Bs. */
                CASE
                    WHEN COUNT(pcd.id) > 0
                        THEN SUM(IFNULL(pcd.monto_bs, 0))

                    WHEN
                        UPPER(
                            LEFT(
                                TRIM(pc.moneda COLLATE utf8mb4_unicode_ci),
                                3
                            )
                        )
                        =
                        _utf8mb4'BS.' COLLATE utf8mb4_unicode_ci
                    THEN IFNULL(pc.pago, 0)

                    ELSE
                        IFNULL(pc.pago, 0)
                        *
                        IFNULL(
                            NULLIF(pc.tasa_cambio, 0),
                            1
                        )
                END AS pagado_bs,

                /* Pago equivalente en USD */
                CASE
                    WHEN COUNT(pcd.id) > 0
                    THEN
                        SUM(
                            CASE
                                WHEN IFNULL(pcd.monto_usd, 0) > 0
                                    THEN pcd.monto_usd

                                WHEN IFNULL(pcd.monto_bs, 0) > 0
                                     AND IFNULL(pcd.tasa_usd, 0) > 0
                                    THEN pcd.monto_bs / pcd.tasa_usd

                                WHEN
                                    UPPER(
                                        LEFT(
                                            TRIM(
                                                pcd.moneda
                                                COLLATE utf8mb4_unicode_ci
                                            ),
                                            3
                                        )
                                    )
                                    <>
                                    _utf8mb4'BS.'
                                    COLLATE utf8mb4_unicode_ci
                                THEN IFNULL(pcd.monto_moneda, 0)

                                ELSE 0
                            END
                        )

                    WHEN
                        UPPER(
                            LEFT(
                                TRIM(pc.moneda COLLATE utf8mb4_unicode_ci),
                                3
                            )
                        )
                        =
                        _utf8mb4'BS.' COLLATE utf8mb4_unicode_ci

                        AND IFNULL(pc.tasa_cambio, 0) > 0
                    THEN pc.pago / pc.tasa_cambio

                    ELSE IFNULL(pc.pago, 0)
                END AS pagado_usd

            FROM pagos_compras AS pc

            LEFT JOIN pagos_compras_detalle AS pcd
                ON pcd.pagos_compras = pc.id

            WHERE
                pc.tipo_documento COLLATE utf8mb4_unicode_ci
                IN (
                    _utf8mb4'GASTO' COLLATE utf8mb4_unicode_ci,
                    _utf8mb4'GASTOS' COLLATE utf8mb4_unicode_ci
                )

            GROUP BY
                pc.id,
                pc.id_documento,
                pc.fecha,
                pc.pago,
                pc.moneda,
                pc.tasa_cambio

        ) AS pago_documento

        GROUP BY
            pago_documento.id_documento

    ) AS pagos
        ON pagos.id_documento = c.id

    WHERE
        c.tipo_documento COLLATE utf8mb4_unicode_ci
        IN (
            _utf8mb4'FC' COLLATE utf8mb4_unicode_ci,
            _utf8mb4'NC' COLLATE utf8mb4_unicode_ci,
            _utf8mb4'ND' COLLATE utf8mb4_unicode_ci,
            _utf8mb4'RC' COLLATE utf8mb4_unicode_ci
        )

) AS base;


/* ============================================================
   2. CUENTAS POR COBRAR
   ============================================================ */

CREATE VIEW view_cuentas_por_cobrar AS
SELECT
    base.id,
    base.cliente,
    base.cliente_rif,
    base.cliente_nombre,

    base.tipo_documento_fiscal,
    base.nro_documento,
    base.nro_control,
    base.fecha,
    base.fecha_documento,
    base.fecha_vencimiento,

    base.moneda,
    base.tasa_dia,
    base.dias_credito,
    base.entregado,
    base.pagado,
    base.doc_afectado,
    base.doc_afe,

    base.igtf,
    base.monto_igtf_bs,

    base.signo_documento,

    /* Monto sin signo */
    base.monto_documento_moneda,
    base.monto_documento_bs,
    base.monto_documento_usd,

    /* Monto con efecto fiscal */
    ROUND(
        base.monto_documento_bs * base.signo_documento,
        2
    ) AS monto_aplicado_bs,

    ROUND(
        base.monto_documento_usd * base.signo_documento,
        2
    ) AS monto_aplicado_usd,

    /* Cobros */
    IFNULL(base.total_cobrado_bs, 0) AS total_cobrado_bs,
    IFNULL(base.total_cobrado_usd, 0) AS total_cobrado_usd,
    IFNULL(base.cantidad_cobros, 0) AS cantidad_cobros,
    base.fecha_ultimo_cobro,

    /* Saldos */
    ROUND(
        (base.monto_documento_bs * base.signo_documento)
        - IFNULL(base.total_cobrado_bs, 0),
        2
    ) AS saldo_bs,

    ROUND(
        (base.monto_documento_usd * base.signo_documento)
        - IFNULL(base.total_cobrado_usd, 0),
        2
    ) AS saldo_usd,

    /* Estado cuenta */
    (
        CASE
            WHEN
                base.tipo_documento_fiscal
                    COLLATE utf8mb4_unicode_ci
                =
                _utf8mb4'NC' COLLATE utf8mb4_unicode_ci
            THEN _utf8mb4'NOTA DE CRÉDITO'

            WHEN ROUND(
                (base.monto_documento_bs * base.signo_documento)
                - IFNULL(base.total_cobrado_bs, 0),
                2
            ) <= 0.01
            THEN _utf8mb4'COBRADO'

            WHEN IFNULL(base.total_cobrado_bs, 0) > 0
            THEN _utf8mb4'COBRO PARCIAL'

            ELSE _utf8mb4'PENDIENTE'
        END
    ) COLLATE utf8mb4_unicode_ci AS estado_cuenta,

    DATEDIFF(
        CURDATE(),
        base.fecha_vencimiento
    ) AS dias_vencido,

    /* Antigüedad */
    (
        CASE
            WHEN
                base.tipo_documento_fiscal
                    COLLATE utf8mb4_unicode_ci
                =
                _utf8mb4'NC' COLLATE utf8mb4_unicode_ci
            THEN _utf8mb4'NOTA DE CRÉDITO'

            WHEN CURDATE() <= base.fecha_vencimiento
            THEN _utf8mb4'NO VENCIDO'

            WHEN DATEDIFF(
                CURDATE(),
                base.fecha_vencimiento
            ) BETWEEN 1 AND 30
            THEN _utf8mb4'1-30 DÍAS'

            WHEN DATEDIFF(
                CURDATE(),
                base.fecha_vencimiento
            ) BETWEEN 31 AND 60
            THEN _utf8mb4'31-60 DÍAS'

            WHEN DATEDIFF(
                CURDATE(),
                base.fecha_vencimiento
            ) BETWEEN 61 AND 90
            THEN _utf8mb4'61-90 DÍAS'

            ELSE _utf8mb4'MÁS DE 90 DÍAS'
        END
    ) COLLATE utf8mb4_unicode_ci AS antiguedad

FROM
(
    SELECT
        s.id,
        s.cliente,

        /* RIF del cliente blindado contra mezcla de collations */
        COALESCE(
            NULLIF(
                s.cliente_ci_rif COLLATE utf8mb4_unicode_ci,
                _utf8mb4'' COLLATE utf8mb4_unicode_ci
            ),
            cl.ci_rif COLLATE utf8mb4_unicode_ci
        ) COLLATE utf8mb4_unicode_ci AS cliente_rif,

        /* Nombre del cliente blindado */
        COALESCE(
            NULLIF(
                s.cliente_nombre COLLATE utf8mb4_unicode_ci,
                _utf8mb4'' COLLATE utf8mb4_unicode_ci
            ),
            cl.nombre COLLATE utf8mb4_unicode_ci
        ) COLLATE utf8mb4_unicode_ci AS cliente_nombre,

        s.documento COLLATE utf8mb4_unicode_ci
            AS tipo_documento_fiscal,

        s.nro_documento COLLATE utf8mb4_unicode_ci
            AS nro_documento,

        s.nro_control COLLATE utf8mb4_unicode_ci
            AS nro_control,

        s.fecha,

        DATE(s.fecha) AS fecha_documento,

        DATE_ADD(
            DATE(s.fecha),
            INTERVAL IFNULL(s.dias_credito, 0) DAY
        ) AS fecha_vencimiento,

        s.moneda COLLATE utf8mb4_unicode_ci
            AS moneda,

        IFNULL(
            NULLIF(s.tasa_dia, 0),
            1
        ) AS tasa_dia,

        IFNULL(s.dias_credito, 0) AS dias_credito,

        s.entregado COLLATE utf8mb4_unicode_ci
            AS entregado,

        s.pagado COLLATE utf8mb4_unicode_ci
            AS pagado,

        s.doc_afectado COLLATE utf8mb4_unicode_ci
            AS doc_afectado,

        s.doc_afe,

        s.igtf COLLATE utf8mb4_unicode_ci
            AS igtf,

        CASE
            WHEN
                s.igtf COLLATE utf8mb4_unicode_ci
                =
                _utf8mb4'S' COLLATE utf8mb4_unicode_ci
            THEN IFNULL(s.monto_igtf, 0)
            ELSE 0
        END AS monto_igtf_bs,

        CASE
            WHEN
                s.documento COLLATE utf8mb4_unicode_ci
                =
                _utf8mb4'NC' COLLATE utf8mb4_unicode_ci
            THEN -1
            ELSE 1
        END AS signo_documento,

        /* ----------------------------------------------------
           Monto en moneda original
           ---------------------------------------------------- */

        CASE
            WHEN
                UPPER(
                    LEFT(
                        TRIM(s.moneda COLLATE utf8mb4_unicode_ci),
                        3
                    )
                )
                =
                _utf8mb4'BS.' COLLATE utf8mb4_unicode_ci
            THEN
                IFNULL(s.total, 0)
                +
                CASE
                    WHEN
                        s.igtf COLLATE utf8mb4_unicode_ci
                        =
                        _utf8mb4'S' COLLATE utf8mb4_unicode_ci
                    THEN IFNULL(s.monto_igtf, 0)
                    ELSE 0
                END

            ELSE
                IFNULL(s.total, 0)
                +
                CASE
                    WHEN
                        s.igtf COLLATE utf8mb4_unicode_ci
                        =
                        _utf8mb4'S' COLLATE utf8mb4_unicode_ci
                        AND IFNULL(s.tasa_dia, 0) > 0
                    THEN s.monto_igtf / s.tasa_dia
                    ELSE 0
                END
        END AS monto_documento_moneda,

        /* ----------------------------------------------------
           Monto expresado en Bs.
           ---------------------------------------------------- */

        CASE
            WHEN
                UPPER(
                    LEFT(
                        TRIM(s.moneda COLLATE utf8mb4_unicode_ci),
                        3
                    )
                )
                =
                _utf8mb4'BS.' COLLATE utf8mb4_unicode_ci
            THEN
                IFNULL(s.total, 0)
                +
                CASE
                    WHEN
                        s.igtf COLLATE utf8mb4_unicode_ci
                        =
                        _utf8mb4'S' COLLATE utf8mb4_unicode_ci
                    THEN IFNULL(s.monto_igtf, 0)
                    ELSE 0
                END

            ELSE
                (
                    IFNULL(s.total, 0)
                    *
                    IFNULL(
                        NULLIF(s.tasa_dia, 0),
                        1
                    )
                )
                +
                CASE
                    WHEN
                        s.igtf COLLATE utf8mb4_unicode_ci
                        =
                        _utf8mb4'S' COLLATE utf8mb4_unicode_ci
                    THEN IFNULL(s.monto_igtf, 0)
                    ELSE 0
                END
        END AS monto_documento_bs,

        /* ----------------------------------------------------
           Monto equivalente en USD
           ---------------------------------------------------- */

        CASE
            WHEN
                UPPER(
                    LEFT(
                        TRIM(s.moneda COLLATE utf8mb4_unicode_ci),
                        3
                    )
                )
                =
                _utf8mb4'BS.' COLLATE utf8mb4_unicode_ci

                AND IFNULL(s.tasa_dia, 0) > 0
            THEN
                (
                    IFNULL(s.total, 0)
                    +
                    CASE
                        WHEN
                            s.igtf COLLATE utf8mb4_unicode_ci
                            =
                            _utf8mb4'S'
                            COLLATE utf8mb4_unicode_ci
                        THEN IFNULL(s.monto_igtf, 0)
                        ELSE 0
                    END
                )
                / s.tasa_dia

            ELSE
                IFNULL(s.total, 0)
                +
                CASE
                    WHEN
                        s.igtf COLLATE utf8mb4_unicode_ci
                        =
                        _utf8mb4'S'
                        COLLATE utf8mb4_unicode_ci

                        AND IFNULL(s.tasa_dia, 0) > 0
                    THEN s.monto_igtf / s.tasa_dia
                    ELSE 0
                END
        END AS monto_documento_usd,

        cobros.total_cobrado_bs,
        cobros.total_cobrado_usd,
        cobros.cantidad_cobros,
        cobros.fecha_ultimo_cobro

    FROM salidas AS s

    LEFT JOIN cliente AS cl
        ON cl.id = s.cliente

    LEFT JOIN
    (
        SELECT
            cobro_documento.id_documento,

            ROUND(
                SUM(cobro_documento.cobrado_bs),
                2
            ) AS total_cobrado_bs,

            ROUND(
                SUM(cobro_documento.cobrado_usd),
                2
            ) AS total_cobrado_usd,

            COUNT(*) AS cantidad_cobros,

            MAX(cobro_documento.fecha) AS fecha_ultimo_cobro

        FROM
        (
            SELECT
                cc.id,
                cc.id_documento,
                cc.fecha,

                /* Cobro en Bs. */
                CASE
                    WHEN COUNT(ccd.id) > 0
                    THEN SUM(IFNULL(ccd.monto_bs, 0))

                    WHEN
                        UPPER(
                            LEFT(
                                TRIM(cc.moneda COLLATE utf8mb4_unicode_ci),
                                3
                            )
                        )
                        =
                        _utf8mb4'BS.' COLLATE utf8mb4_unicode_ci
                    THEN IFNULL(cc.pago, 0)

                    ELSE
                        IFNULL(cc.pago, 0)
                        *
                        IFNULL(
                            NULLIF(cc.tasa_cambio, 0),
                            1
                        )
                END AS cobrado_bs,

                /* Cobro equivalente USD */
                CASE
                    WHEN COUNT(ccd.id) > 0
                    THEN
                        SUM(
                            CASE
                                WHEN IFNULL(ccd.monto_usd, 0) > 0
                                THEN ccd.monto_usd

                                WHEN IFNULL(ccd.monto_bs, 0) > 0
                                     AND IFNULL(ccd.tasa_usd, 0) > 0
                                THEN ccd.monto_bs / ccd.tasa_usd

                                WHEN
                                    UPPER(
                                        LEFT(
                                            TRIM(
                                                ccd.moneda
                                                COLLATE utf8mb4_unicode_ci
                                            ),
                                            3
                                        )
                                    )
                                    <>
                                    _utf8mb4'BS.'
                                    COLLATE utf8mb4_unicode_ci
                                THEN IFNULL(ccd.monto_moneda, 0)

                                ELSE 0
                            END
                        )

                    WHEN
                        UPPER(
                            LEFT(
                                TRIM(cc.moneda COLLATE utf8mb4_unicode_ci),
                                3
                            )
                        )
                        =
                        _utf8mb4'BS.' COLLATE utf8mb4_unicode_ci

                        AND IFNULL(cc.tasa_cambio, 0) > 0
                    THEN cc.pago / cc.tasa_cambio

                    ELSE IFNULL(cc.pago, 0)
                END AS cobrado_usd

            FROM cobros_cliente AS cc

            LEFT JOIN cobros_cliente_detalle AS ccd
                ON ccd.cobros_cliente = cc.id

            GROUP BY
                cc.id,
                cc.id_documento,
                cc.fecha,
                cc.pago,
                cc.moneda,
                cc.tasa_cambio

        ) AS cobro_documento

        GROUP BY
            cobro_documento.id_documento

    ) AS cobros
        ON cobros.id_documento = s.id

    WHERE
        s.tipo_documento COLLATE utf8mb4_unicode_ci
            =
        _utf8mb4'TDCFCV' COLLATE utf8mb4_unicode_ci

        AND
        s.estatus COLLATE utf8mb4_unicode_ci
            =
        _utf8mb4'PROCESADO' COLLATE utf8mb4_unicode_ci

        AND
        s.documento COLLATE utf8mb4_unicode_ci
        IN (
            _utf8mb4'FC' COLLATE utf8mb4_unicode_ci,
            _utf8mb4'NC' COLLATE utf8mb4_unicode_ci,
            _utf8mb4'ND' COLLATE utf8mb4_unicode_ci,
            _utf8mb4'RC' COLLATE utf8mb4_unicode_ci
        )

) AS base;


/* ============================================================
   3. RESUMEN CUENTAS POR PAGAR
   ============================================================ */

CREATE VIEW view_cuentas_por_pagar_resumen AS
SELECT
    proveedor,

    proveedor_rif COLLATE utf8mb4_unicode_ci
        AS proveedor_rif,

    proveedor_nombre COLLATE utf8mb4_unicode_ci
        AS proveedor_nombre,

    COUNT(*) AS cantidad_documentos,

    SUM(
        CASE
            WHEN
                estado_cuenta COLLATE utf8mb4_unicode_ci
                =
                _utf8mb4'PENDIENTE' COLLATE utf8mb4_unicode_ci
            THEN 1
            ELSE 0
        END
    ) AS documentos_pendientes,

    SUM(
        CASE
            WHEN
                estado_cuenta COLLATE utf8mb4_unicode_ci
                =
                _utf8mb4'PAGO PARCIAL'
                COLLATE utf8mb4_unicode_ci
            THEN 1
            ELSE 0
        END
    ) AS documentos_parciales,

    ROUND(
        SUM(monto_aplicado_bs),
        2
    ) AS monto_documentos_bs,

    ROUND(
        SUM(monto_aplicado_usd),
        2
    ) AS monto_documentos_usd,

    ROUND(
        SUM(total_pagado_bs),
        2
    ) AS total_pagado_bs,

    ROUND(
        SUM(total_pagado_usd),
        2
    ) AS total_pagado_usd,

    ROUND(
        SUM(saldo_bs),
        2
    ) AS saldo_bs,

    ROUND(
        SUM(saldo_usd),
        2
    ) AS saldo_usd

FROM view_cuentas_por_pagar

WHERE
    anulado COLLATE utf8mb4_unicode_ci
    =
    _utf8mb4'N' COLLATE utf8mb4_unicode_ci

GROUP BY
    proveedor,
    proveedor_rif COLLATE utf8mb4_unicode_ci,
    proveedor_nombre COLLATE utf8mb4_unicode_ci;
